cmake_minimum_required(VERSION 3.22)
project(espaperplay_host C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
option(EPP_SANITIZERS "Enable host AddressSanitizer and UndefinedBehaviorSanitizer" ON)
add_compile_options(-Wall -Wextra -Werror -Wpedantic)
if(EPP_SANITIZERS AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
add_library(epp_runtime STATIC ../components/epp_runtime/src/epp_runtime.c)
target_include_directories(epp_runtime PUBLIC ../components/epp_runtime/include)
add_library(epp_demo STATIC ../components/epp_demo/epp_demo.c)
target_include_directories(epp_demo PUBLIC ../components/epp_demo/include)
target_link_libraries(epp_demo PUBLIC epp_runtime)
add_executable(runtime_tests runtime/test_runtime.c)
target_link_libraries(runtime_tests PRIVATE epp_runtime)
add_test(NAME runtime COMMAND runtime_tests)
add_executable(bootstrap_demo bootstrap_main.c)
target_link_libraries(bootstrap_demo PRIVATE epp_demo)
add_test(NAME bootstrap COMMAND bootstrap_demo)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME board_profile COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../tools/check_board.py" "${CMAKE_CURRENT_SOURCE_DIR}/../boards/legacy_n16r8.json")
add_test(NAME partition_layout COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../tools/check_layout.py" "${CMAKE_CURRENT_SOURCE_DIR}/../partitions.csv")
add_test(NAME config_rejections COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_config.py")
set_tests_properties(runtime bootstrap PROPERTIES ENVIRONMENT "ASAN_OPTIONS=detect_leaks=1;UBSAN_OPTIONS=halt_on_error=1")
